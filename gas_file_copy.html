H列が「○」のファイルを、K列で指定したGoogle Driveのフォルダへコピーし、L列の名前へリネームするGASコードです。コピー先はフォルダID・URL・マイドライブのパスに対応します。

function copyMarkedFiles() {

  // ==========================================
  // 設定
  // ==========================================
  const START_ROW = 2;

  const COL_TYPE       = 3;  // C列：種類
  const COL_NAME       = 4;  // D列：元ファイル名
  const COL_FILE_ID    = 5;  // E列：ファイルID
  const COL_MARK       = 8;  // H列：○
  const COL_DEST       = 11; // K列：コピー先
  const COL_NEW_NAME   = 12; // L列：コピー後ファイル名


  // ==========================================
  // コントロール側スプレッドシート
  // ==========================================
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getActiveSheet();

  const lastRow = sheet.getLastRow();

  if (lastRow < START_ROW) {
    return;
  }


  // ==========================================
  // C列～L列をまとめて取得
  // ==========================================
  const data = sheet
    .getRange(
      START_ROW,
      COL_TYPE,
      lastRow - START_ROW + 1,
      COL_NEW_NAME - COL_TYPE + 1
    )
    .getDisplayValues();


  // ==========================================
  // 1行ずつ処理
  // ==========================================
  data.forEach(function(row, index) {

    const sheetRow = START_ROW + index;

    // C列
    const type = row[0].trim();

    // D列
    const originalName = row[1].trim();

    // E列
    const fileId = row[2].trim();

    // H列
    const mark = row[5].trim();

    // K列
    const destination = row[8].trim();

    // L列
    const newFileName = row[9].trim();


    // ------------------------------------------
    // H列が「○」でなければ何もしない
    // ------------------------------------------
    if (mark !== "○") {
      return;
    }


    // ------------------------------------------
    // フォルダはコピー対象外
    // ------------------------------------------
    if (type !== "ファイル") {

      Logger.log(
        "行 " + sheetRow +
        "：ファイルではないためスキップ"
      );

      return;
    }


    // ------------------------------------------
    // ファイルIDチェック
    // ------------------------------------------
    if (fileId === "") {

      Logger.log(
        "行 " + sheetRow +
        "：ファイルIDがありません"
      );

      return;
    }


    // ------------------------------------------
    // コピー先チェック
    // ------------------------------------------
    if (destination === "") {

      Logger.log(
        "行 " + sheetRow +
        "：コピー先がありません"
      );

      return;
    }


    // ------------------------------------------
    // 新しいファイル名チェック
    // ------------------------------------------
    if (newFileName === "") {

      Logger.log(
        "行 " + sheetRow +
        "：コピー後ファイル名がありません"
      );

      return;
    }


    try {

      // ========================================
      // 元ファイルを取得
      // ========================================
      const sourceFile =
        DriveApp.getFileById(fileId);


      // ========================================
      // コピー先フォルダを取得
      // ========================================
      const destinationFolder =
        getDestinationFolder(destination);


      // ========================================
      // コピー ＋ リネーム
      // ========================================
      const copiedFile =
        sourceFile.makeCopy(
          newFileName,
          destinationFolder
        );


      Logger.log(
        "コピー成功：" +
        originalName +
        " → " +
        copiedFile.getName()
      );


    } catch (error) {

      Logger.log(
        "行 " +
        sheetRow +
        " エラー：" +
        error.message
      );
    }

  });


  SpreadsheetApp.flush();
}



/**
 * K列の内容からコピー先フォルダを取得
 *
 * 次の3種類に対応
 *
 * 1. フォルダID
 * 2. Google DriveのフォルダURL
 * 3. マイドライブからのパス
 *
 * 例：
 * 1-apFeDvDwJ6euzwD...
 *
 * https://drive.google.com/drive/folders/1-apFeDvDwJ6euzwD...
 *
 * マイドライブ/作業/20260831
 *
 */
function getDestinationFolder(destination) {

  destination = destination.trim();


  // ==========================================
  // Google DriveのフォルダURLの場合
  // ==========================================
  if (
    destination.indexOf(
      "drive.google.com"
    ) !== -1
  ) {

    const match =
      destination.match(
        /\/folders\/([a-zA-Z0-9_-]+)/
      );

    if (!match) {
      throw new Error(
        "コピー先フォルダURLを認識できません"
      );
    }

    return DriveApp.getFolderById(
      match[1]
    );
  }


  // ==========================================
  // フォルダIDの可能性がある場合
  // ==========================================
  if (
    destination.indexOf("/") === -1 &&
    destination.length >= 15
  ) {

    try {

      return DriveApp.getFolderById(
        destination
      );

    } catch (error) {

      // IDではなかった場合は
      // 下のパス検索へ進む

    }
  }


  // ==========================================
  // マイドライブのパスとして処理
  // ==========================================

  let path = destination;

  // 「マイドライブ/」を除去
  path = path.replace(
    /^マイドライブ\/?/,
    ""
  );

  // 前後の / を除去
  path = path.replace(
    /^\/+|\/+$/g,
    ""
  );


  // マイドライブ直下
  if (path === "") {
    return DriveApp.getRootFolder();
  }


  const folderNames =
    path
      .split("/")
      .filter(function(name) {
        return name.trim() !== "";
      });


  let currentFolder =
    DriveApp.getRootFolder();


  // ==========================================
  // 1階層ずつフォルダを探す
  // ==========================================
  folderNames.forEach(function(folderName) {

    folderName = folderName.trim();

    const folders =
      currentFolder.getFoldersByName(
        folderName
      );


    if (!folders.hasNext()) {

      throw new Error(
        "コピー先フォルダが見つかりません：" +
        folderName
      );
    }


    const foundFolder =
      folders.next();


    // 同じ階層に同名フォルダが複数ある場合
    if (folders.hasNext()) {

      throw new Error(
        "同じ名前のフォルダが複数あります：" +
        folderName
      );
    }


    currentFolder = foundFolder;

  });


  return currentFolder;
}
