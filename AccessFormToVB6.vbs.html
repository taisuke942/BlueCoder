AccessFormToVB6.vbs

Accessフォームを読み取り、VB6用の.frmファイルを 作成するVBSコードです。 
Option Explicit

' ============================================================
' AccessフォームをVB6の.frmファイルへ変換する簡易ツール
'
' 使い方
' 1. このVBSをダブルクリック
' 2. Accessファイル（.accdb / .mdb）を選択
' 3. 同じ場所に「VB6_Forms」フォルダが作成される
' 4. その中の.frmファイルをVB6へ追加する
'
' 注意
' ・画面の位置・大きさ・基本コントロールを移すための「ひな形」です。
' ・Access VBA、RecordSource、ControlSource、サブフォームの動作は
'   自動移植されません。
' ============================================================

Const acDesign = 1
Const acForm = 2
Const acSaveNo = 2
Const msoFileDialogFilePicker = 3

Dim acc
Dim fso
Dim picker
Dim dbPath
Dim dbFolder
Dim outputFolder
Dim logPath
Dim logFile
Dim formCount
Dim i
Dim formName
Dim successCount
Dim failedCount
Dim resultMessage

Set fso = CreateObject("Scripting.FileSystemObject")
Set acc = CreateObject("Access.Application")

On Error Resume Next

' Accessファイルを選択
Set picker = acc.FileDialog(msoFileDialogFilePicker)

If Err.Number <> 0 Then
    MsgBox "Accessのファイル選択画面を開けませんでした。" & vbCrLf & _
           "Microsoft Accessがインストールされているか確認してください。", _
           vbCritical, "Accessフォーム変換"
    acc.Quit
    WScript.Quit
End If

On Error GoTo 0

With picker
    .Title = "変換するAccessファイルを選択してください"
    .AllowMultiSelect = False
    .Filters.Clear
    .Filters.Add "Accessファイル", "*.accdb;*.mdb"

    If .Show <> -1 Then
        acc.Quit
        WScript.Quit
    End If

    dbPath = .SelectedItems(1)
End With

dbFolder = fso.GetParentFolderName(dbPath)
outputFolder = fso.BuildPath(dbFolder, "VB6_Forms")

If Not fso.FolderExists(outputFolder) Then
    fso.CreateFolder outputFolder
End If

logPath = fso.BuildPath(outputFolder, "変換結果.txt")
Set logFile = fso.CreateTextFile(logPath, True, True)

logFile.WriteLine "Accessフォーム → VB6フォーム 変換結果"
logFile.WriteLine "Accessファイル: " & dbPath
logFile.WriteLine "作成日時: " & Now
logFile.WriteLine String(60, "-")

On Error Resume Next
acc.OpenCurrentDatabase dbPath

If Err.Number <> 0 Then
    MsgBox "Accessファイルを開けませんでした。" & vbCrLf & _
           Err.Description, vbCritical, "Accessフォーム変換"
    logFile.WriteLine "エラー: " & Err.Description
    logFile.Close
    acc.Quit
    WScript.Quit
End If

On Error GoTo 0

formCount = acc.CurrentProject.AllForms.Count

If formCount = 0 Then
    MsgBox "このAccessファイルにはフォームがありません。", _
           vbInformation, "Accessフォーム変換"
    logFile.WriteLine "フォームはありませんでした。"
    logFile.Close
    acc.CloseCurrentDatabase
    acc.Quit
    WScript.Quit
End If

successCount = 0
failedCount = 0

For i = 0 To formCount - 1

    formName = acc.CurrentProject.AllForms(i).Name

    If ConvertOneForm(acc, fso, formName, outputFolder, logFile) Then
        successCount = successCount + 1
    Else
        failedCount = failedCount + 1
    End If

Next

logFile.WriteLine String(60, "-")
logFile.WriteLine "成功: " & successCount & "フォーム"
logFile.WriteLine "失敗: " & failedCount & "フォーム"
logFile.Close

acc.CloseCurrentDatabase
acc.Quit

Set picker = Nothing
Set acc = Nothing
Set logFile = Nothing
Set fso = Nothing

resultMessage = _
    "変換が完了しました。" & vbCrLf & vbCrLf & _
    "成功: " & successCount & "フォーム" & vbCrLf & _
    "失敗: " & failedCount & "フォーム" & vbCrLf & vbCrLf & _
    "作成場所:" & vbCrLf & outputFolder & vbCrLf & vbCrLf & _
    "この中の.frmファイルをVB6へ追加してください。"

MsgBox resultMessage, vbInformation, "Accessフォーム変換"


Function ConvertOneForm(ByRef accessApp, ByRef fileSystem, _
                        ByVal sourceFormName, ByVal destinationFolder, _
                        ByRef resultLog)

    Dim frm
    Dim ctl
    Dim outputFile
    Dim outputPath
    Dim vbFormName
    Dim vbControlName
    Dim controlClass
    Dim captionText
    Dim formCaption
    Dim formWidth
    Dim formHeight
    Dim headerHeight
    Dim detailHeight
    Dim footerHeight
    Dim topOffset
    Dim tabIndex
    Dim controlSource
    Dim rowSource
    Dim mappingText
    Dim unsupportedCount
    Dim errText

    On Error Resume Next

    accessApp.DoCmd.OpenForm sourceFormName, acDesign

    If Err.Number <> 0 Then
        errText = Err.Description
        Err.Clear

        resultLog.WriteLine "失敗: " & sourceFormName & " / " & errText
        ConvertOneForm = False
        Exit Function
    End If

    Set frm = accessApp.Forms(sourceFormName)

    If Err.Number <> 0 Or frm Is Nothing Then
        errText = Err.Description
        Err.Clear

        resultLog.WriteLine "失敗: " & sourceFormName & " / " & errText
        accessApp.DoCmd.Close acForm, sourceFormName, acSaveNo
        ConvertOneForm = False
        Exit Function
    End If

    vbFormName = MakeVbName(sourceFormName, "frm")

    If LCase(Left(vbFormName, 3)) <> "frm" Then
        vbFormName = "frm" & vbFormName
    End If

    outputPath = fileSystem.BuildPath( _
        destinationFolder, vbFormName & ".frm")

    Set outputFile = fileSystem.CreateTextFile(outputPath, True, False)

    If Err.Number <> 0 Then
        errText = Err.Description
        Err.Clear

        resultLog.WriteLine "失敗: " & sourceFormName & " / " & errText
        accessApp.DoCmd.Close acForm, sourceFormName, acSaveNo
        ConvertOneForm = False
        Exit Function
    End If

    formCaption = GetPropertyText(frm, "Caption")
    If formCaption = "" Then
        formCaption = sourceFormName
    End If

    formWidth = GetNumericProperty(frm, "InsideWidth", 9000)

    headerHeight = GetSectionHeight(frm, 1)
    detailHeight = GetSectionHeight(frm, 0)
    footerHeight = GetSectionHeight(frm, 2)

    formHeight = headerHeight + detailHeight + footerHeight

    If formHeight <= 0 Then
        formHeight = 6000
    End If

    outputFile.WriteLine "VERSION 5.00"
    outputFile.WriteLine "Begin VB.Form " & vbFormName
    outputFile.WriteLine "   Caption         =   """ & EscapeVbString(formCaption) & """"
    outputFile.WriteLine "   ClientHeight    =   " & CStr(formHeight)
    outputFile.WriteLine "   ClientLeft      =   60"
    outputFile.WriteLine "   ClientTop       =   345"
    outputFile.WriteLine "   ClientWidth     =   " & CStr(formWidth)
    outputFile.WriteLine "   LinkTopic       =   ""Form1"""
    outputFile.WriteLine "   ScaleHeight     =   " & CStr(formHeight)
    outputFile.WriteLine "   ScaleWidth      =   " & CStr(formWidth)
    outputFile.WriteLine "   StartUpPosition =   3  'Windows の既定値"

    tabIndex = 0
    mappingText = ""
    unsupportedCount = 0

    For Each ctl In frm.Controls

        controlClass = GetVb6ControlClass(GetNumericProperty(ctl, "ControlType", -1))
        vbControlName = MakeVbName(GetPropertyText(ctl, "Name"), "ctl")

        If controlClass = "" Then

            unsupportedCount = unsupportedCount + 1

            ' 未対応コントロールは、位置確認用のラベルに置き換える
            outputFile.WriteLine "   Begin VB.Label " & vbControlName
            outputFile.WriteLine "      Caption         =   ""[未対応] " & _
                                 EscapeVbString(GetPropertyText(ctl, "Name")) & """"
            outputFile.WriteLine "      Height          =   " & _
                                 CStr(GetNumericProperty(ctl, "Height", 315))
            outputFile.WriteLine "      Left            =   " & _
                                 CStr(GetNumericProperty(ctl, "Left", 0))
            outputFile.WriteLine "      Top             =   " & _
                                 CStr(GetControlTop(frm, ctl, headerHeight, detailHeight))
            outputFile.WriteLine "      Width           =   " & _
                                 CStr(GetNumericProperty(ctl, "Width", 1200))
            outputFile.WriteLine "   End"

        Else

            outputFile.WriteLine "   Begin " & controlClass & " " & vbControlName

            captionText = GetPropertyText(ctl, "Caption")

            If HasCaption(controlClass) Then
                If captionText = "" Then
                    captionText = GetPropertyText(ctl, "Name")
                End If

                outputFile.WriteLine "      Caption         =   """ & _
                                     EscapeVbString(captionText) & """"
            End If

            outputFile.WriteLine "      Height          =   " & _
                                 CStr(GetNumericProperty(ctl, "Height", 315))
            outputFile.WriteLine "      Left            =   " & _
                                 CStr(GetNumericProperty(ctl, "Left", 0))

            If HasTabIndex(controlClass) Then
                outputFile.WriteLine "      TabIndex        =   " & CStr(tabIndex)
                tabIndex = tabIndex + 1
            End If

            topOffset = GetControlTop(frm, ctl, headerHeight, detailHeight)

            outputFile.WriteLine "      Top             =   " & CStr(topOffset)
            outputFile.WriteLine "      Width           =   " & _
                                 CStr(GetNumericProperty(ctl, "Width", 1200))
            outputFile.WriteLine "   End"

        End If

        controlSource = GetPropertyText(ctl, "ControlSource")
        rowSource = GetPropertyText(ctl, "RowSource")

        If controlSource <> "" Then
            mappingText = mappingText & _
                          "' " & vbControlName & _
                          "  ← Access ControlSource: " & _
                          Replace(controlSource, vbCr, " ") & vbCrLf
        End If

        If rowSource <> "" Then
            mappingText = mappingText & _
                          "' " & vbControlName & _
                          "  ← Access RowSource: " & _
                          Replace(rowSource, vbCr, " ") & vbCrLf
        End If

    Next

    outputFile.WriteLine "End"
    outputFile.WriteLine "Attribute VB_Name = """ & vbFormName & """"
    outputFile.WriteLine "Attribute VB_GlobalNameSpace = False"
    outputFile.WriteLine "Attribute VB_Creatable = False"
    outputFile.WriteLine "Attribute VB_PredeclaredId = True"
    outputFile.WriteLine "Attribute VB_Exposed = False"
    outputFile.WriteLine "Option Explicit"
    outputFile.WriteLine ""
    outputFile.WriteLine "' ------------------------------------------------------------"
    outputFile.WriteLine "' Accessから引き継いだ情報"
    outputFile.WriteLine "' RecordSource: " & _
                         Replace(GetPropertyText(frm, "RecordSource"), vbCr, " ")

    If mappingText <> "" Then
        outputFile.Write mappingText
    End If

    outputFile.WriteLine "' ------------------------------------------------------------"

    outputFile.Close

    accessApp.DoCmd.Close acForm, sourceFormName, acSaveNo

    resultLog.WriteLine _
        "成功: " & sourceFormName & _
        " → " & vbFormName & ".frm" & _
        " / 未対応コントロール: " & unsupportedCount

    ConvertOneForm = True

    On Error GoTo 0

End Function


Function GetVb6ControlClass(ByVal accessControlType)

    Select Case CLng(accessControlType)

        Case 100
            GetVb6ControlClass = "VB.Label"

        Case 104
            GetVb6ControlClass = "VB.CommandButton"

        Case 105
            GetVb6ControlClass = "VB.OptionButton"

        Case 106
            GetVb6ControlClass = "VB.CheckBox"

        Case 107
            GetVb6ControlClass = "VB.Frame"

        Case 109
            GetVb6ControlClass = "VB.TextBox"

        Case 110
            GetVb6ControlClass = "VB.ListBox"

        Case 111
            GetVb6ControlClass = "VB.ComboBox"

        Case 122
            GetVb6ControlClass = "VB.CommandButton"

        Case Else
            GetVb6ControlClass = ""

    End Select

End Function


Function HasCaption(ByVal vbClassName)

    Select Case vbClassName

        Case "VB.Label", _
             "VB.CommandButton", _
             "VB.OptionButton", _
             "VB.CheckBox", _
             "VB.Frame"

            HasCaption = True

        Case Else
            HasCaption = False

    End Select

End Function


Function HasTabIndex(ByVal vbClassName)

    Select Case vbClassName

        Case "VB.CommandButton", _
             "VB.OptionButton", _
             "VB.CheckBox", _
             "VB.TextBox", _
             "VB.ListBox", _
             "VB.ComboBox"

            HasTabIndex = True

        Case Else
            HasTabIndex = False

    End Select

End Function


Function GetControlTop(ByRef sourceForm, ByRef sourceControl, _
                       ByVal formHeaderHeight, ByVal formDetailHeight)

    Dim sectionNumber
    Dim originalTop

    sectionNumber = GetNumericProperty(sourceControl, "Section", 0)
    originalTop = GetNumericProperty(sourceControl, "Top", 0)

    Select Case sectionNumber

        Case 1
            GetControlTop = originalTop

        Case 0
            GetControlTop = formHeaderHeight + originalTop

        Case 2
            GetControlTop = formHeaderHeight + _
                            formDetailHeight + originalTop

        Case Else
            GetControlTop = originalTop

    End Select

End Function


Function GetSectionHeight(ByRef sourceForm, ByVal sectionNumber)

    Dim value

    On Error Resume Next

    Err.Clear
    value = sourceForm.Section(sectionNumber).Height

    If Err.Number <> 0 Then
        Err.Clear
        value = 0
    End If

    On Error GoTo 0

    GetSectionHeight = CLng(value)

End Function


Function GetPropertyText(ByRef targetObject, ByVal propertyName)

    Dim value

    On Error Resume Next

    Err.Clear
    value = targetObject.Properties(propertyName).Value

    If Err.Number <> 0 Or IsNull(value) Then
        Err.Clear
        value = ""
    End If

    On Error GoTo 0

    GetPropertyText = CStr(value)

End Function


Function GetNumericProperty(ByRef targetObject, ByVal propertyName, _
                            ByVal defaultValue)

    Dim value

    On Error Resume Next

    Err.Clear
    value = targetObject.Properties(propertyName).Value

    If Err.Number <> 0 Or IsNull(value) Or Not IsNumeric(value) Then
        Err.Clear
        value = defaultValue
    End If

    On Error GoTo 0

    GetNumericProperty = CLng(value)

End Function


Function MakeVbName(ByVal originalName, ByVal prefixText)

    Dim result
    Dim i
    Dim ch

    result = ""

    For i = 1 To Len(originalName)

        ch = Mid(originalName, i, 1)

        If IsAllowedNameCharacter(ch) Then
            result = result & ch
        Else
            result = result & "_"
        End If

    Next

    If result = "" Then
        result = prefixText
    End If

    If IsNumeric(Left(result, 1)) Then
        result = prefixText & "_" & result
    End If

    MakeVbName = result

End Function


Function IsAllowedNameCharacter(ByVal oneCharacter)

    Dim code

    code = AscW(oneCharacter)

    If (code >= 48 And code <= 57) Or _
       (code >= 65 And code <= 90) Or _
       (code >= 97 And code <= 122) Or _
       oneCharacter = "_" Or _
       code > 127 Then

        IsAllowedNameCharacter = True
    Else
        IsAllowedNameCharacter = False
    End If

End Function


Function EscapeVbString(ByVal value)

    EscapeVbString = Replace(value, """", """""")

End Function
