H列が「○」のGoogleスプレッドシートを開き、「シート1」のA1の値を取得して、コントロール側のI列へ書き込むGASコードです。

function getA1FromMarkedSpreadsheets() {

  // ==========================================
  // 設定
  // ==========================================
  const START_ROW = 2;

  const COL_FILE_ID   = 5; // E列
  const COL_MIME_TYPE = 7; // G列
  const COL_MARK      = 8; // H列
  const COL_RESULT    = 9; // I列

  const TARGET_SHEET_NAME = "シート1";

  const SPREADSHEET_MIME =
    "application/vnd.google-apps.spreadsheet";


  // ==========================================
  // コントロール側スプレッドシート
  // ==========================================
  const controlSS =
    SpreadsheetApp.getActiveSpreadsheet();

  const controlSheet =
    controlSS.getActiveSheet();


  // 最終行取得
  const lastRow = controlSheet.getLastRow();

  if (lastRow < START_ROW) {
    return;
  }


  // ==========================================
  // E～H列を一括取得
  // ==========================================
  const data = controlSheet
    .getRange(
      START_ROW,
      COL_FILE_ID,
      lastRow - START_ROW + 1,
      4
    )
    .getDisplayValues();


  // ==========================================
  // 1行ずつ処理
  // ==========================================
  data.forEach(function(row, index) {

    const sheetRow = START_ROW + index;

    // E列
    const fileId = row[0].trim();

    // G列
    const mimeType = row[2].trim();

    // H列
    const mark = row[3].trim();


    // ------------------------------------------
    // H列が「○」でなければ何もしない
    // ------------------------------------------
    if (mark !== "○") {
      return;
    }


    // ------------------------------------------
    // Googleスプレッドシート以外は何もしない
    // ------------------------------------------
    if (mimeType !== SPREADSHEET_MIME) {
      return;
    }


    // ------------------------------------------
    // ファイルIDがない場合
    // ------------------------------------------
    if (fileId === "") {

      controlSheet
        .getRange(sheetRow, COL_RESULT)
        .setValue("エラー：ファイルIDなし");

      return;
    }


    try {

      // ========================================
      // 対象スプレッドシートを開く
      // ========================================
      const targetSS =
        SpreadsheetApp.openById(fileId);


      // ========================================
      // 「シート1」を取得
      // ========================================
      const targetSheet =
        targetSS.getSheetByName(
          TARGET_SHEET_NAME
        );


      // シート1が存在しない場合
      if (targetSheet === null) {

        controlSheet
          .getRange(sheetRow, COL_RESULT)
          .setValue("エラー：シート1がありません");

        return;
      }


      // ========================================
      // A1の値を取得
      // ========================================
      const value =
        targetSheet
          .getRange("A1")
          .getValue();


      // ========================================
      // コントロール側のI列へ書き込む
      // ========================================
      controlSheet
        .getRange(sheetRow, COL_RESULT)
        .setValue(value);


    } catch (error) {

      // エラー内容をI列へ表示
      controlSheet
        .getRange(sheetRow, COL_RESULT)
        .setValue(
          "エラー：" + error.message
        );
    }

  });


  SpreadsheetApp.flush();
}
