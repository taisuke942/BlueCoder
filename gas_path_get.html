マイドライブ配下のファイル・フォルダを再帰的に取得し、パス・種類・名前・ID・URL・MIMEタイプをスプレッドシートへ一覧出力するGASコードです。

function getMyDriveFileList() {

  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getActiveSheet();

  // -----------------------------
  // 出力設定
  // -----------------------------
  const START_ROW = 2;
  const START_COL = 2; // B列

  // 見出し
  const headers = [
    "パス",
    "種類",
    "名前",
    "ID",
    "URL",
    "MIMEタイプ"
  ];

  sheet
    .getRange(1, START_COL, 1, headers.length)
    .setValues([headers]);

  // B2:G の既存内容をクリア
  const clearRows = sheet.getMaxRows() - START_ROW + 1;

  if (clearRows > 0) {
    sheet
      .getRange(
        START_ROW,
        START_COL,
        clearRows,
        headers.length
      )
      .clearContent();
  }

  // -----------------------------
  // マイドライブ取得
  // -----------------------------
  const rootFolder = DriveApp.getRootFolder();

  const output = [];

  // マイドライブ配下を取得
  scanFolder(
    rootFolder,
    "マイドライブ",
    output
  );

  // パス順に並び替え
  output.sort(function(a, b) {
    return a[0].localeCompare(b[0], "ja");
  });

  // -----------------------------
  // 行数が不足していたら追加
  // -----------------------------
  const requiredRows =
    START_ROW + output.length - 1;

  if (requiredRows > sheet.getMaxRows()) {

    sheet.insertRowsAfter(
      sheet.getMaxRows(),
      requiredRows - sheet.getMaxRows()
    );
  }

  // -----------------------------
  // 一括出力
  // -----------------------------
  if (output.length > 0) {

    sheet
      .getRange(
        START_ROW,
        START_COL,
        output.length,
        headers.length
      )
      .setValues(output);
  }
}


/**
 * フォルダ内を再帰的に調査する
 *
 * @param {Folder} folder
 * @param {string} currentPath
 * @param {Array} output
 */
function scanFolder(folder, currentPath, output) {

  // =============================
  // このフォルダ内のファイル
  // =============================
  const fileIterator = folder.getFiles();

  const files = [];

  while (fileIterator.hasNext()) {
    files.push(fileIterator.next());
  }

  // ファイル名順
  files.sort(function(a, b) {
    return a.getName().localeCompare(
      b.getName(),
      "ja"
    );
  });

  files.forEach(function(file) {

    const filePath =
      currentPath +
      "/" +
      file.getName();

    output.push([
      filePath,          // B：パス
      "ファイル",        // C：種類
      file.getName(),    // D：名前
      file.getId(),      // E：ID
      file.getUrl(),     // F：URL
      file.getMimeType() // G：MIMEタイプ
    ]);
  });


  // =============================
  // 下位フォルダ
  // =============================
  const folderIterator = folder.getFolders();

  const folders = [];

  while (folderIterator.hasNext()) {
    folders.push(folderIterator.next());
  }

  // フォルダ名順
  folders.sort(function(a, b) {
    return a.getName().localeCompare(
      b.getName(),
      "ja"
    );
  });


  folders.forEach(function(subFolder) {

    const folderPath =
      currentPath +
      "/" +
      subFolder.getName();

    // フォルダ自体も1行出力
    output.push([
      folderPath,          // B：パス
      "フォルダ",          // C：種類
      subFolder.getName(), // D：名前
      subFolder.getId(),   // E：ID
      subFolder.getUrl(),  // F：URL
      ""                   // G：MIMEタイプ
    ]);

    // さらに下を調べる
    scanFolder(
      subFolder,
      folderPath,
      output
    );
  });
}
