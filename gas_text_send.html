H列が「○」のGoogleスプレッドシートを開き、コントロール側J列の値を対象ファイルの「シート1」B1へ書き込むGASコードです。

function pasteJToMarkedSpreadsheets() {

  // ==========================================
  // 設定
  // ==========================================
  const START_ROW = 2;

  const COL_FILE_ID   = 5;  // E列
  const COL_MIME_TYPE = 7;  // G列
  const COL_MARK      = 8;  // H列
  const COL_SOURCE    = 10; // J列

  const TARGET_SHEET_NAME = "シート1";

  const SPREADSHEET_MIME =
    "application/vnd.google-apps.spreadsheet";


  // ==========================================
  // コントロール側スプレッドシート
  // ==========================================
  const controlSS = SpreadsheetApp.getActiveSpreadsheet();
  const controlSheet = controlSS.getActiveSheet();

  const lastRow = controlSheet.getLastRow();

  if (lastRow < START_ROW) {
    return;
  }


  // ==========================================
  // E列～J列をまとめて取得
  // ==========================================
  const data = controlSheet
    .getRange(
      START_ROW,
      COL_FILE_ID,
      lastRow - START_ROW + 1,
      6
    )
    .getValues();


  // ==========================================
  // 1行ずつ処理
  // ==========================================
  data.forEach(function(row) {

    // E列
    const fileId = String(row[0]).trim();

    // G列
    const mimeType = String(row[2]).trim();

    // H列
    const mark = String(row[3]).trim();

    // J列
    const valueToPaste = row[5];


    // ------------------------------------------
    // H列が「○」でなければ処理しない
    // ------------------------------------------
    if (mark !== "○") {
      return;
    }


    // ------------------------------------------
    // Googleスプレッドシート以外は処理しない
    // ------------------------------------------
    if (mimeType !== SPREADSHEET_MIME) {
      return;
    }


    // ------------------------------------------
    // ファイルIDが空なら処理しない
    // ------------------------------------------
    if (fileId === "") {
      return;
    }


    try {

      // 対象のGoogleスプレッドシートを開く
      const targetSS =
        SpreadsheetApp.openById(fileId);


      // 「シート1」を取得
      const targetSheet =
        targetSS.getSheetByName(TARGET_SHEET_NAME);


      // 「シート1」が存在しない場合
      if (targetSheet === null) {
        Logger.log(
          "シート1がありません：" +
          targetSS.getName()
        );
        return;
      }


      // ========================================
      // J列の値を対象ファイルのB1へ貼り付け
      // ========================================
      targetSheet
        .getRange("B1")
        .setValue(valueToPaste);


    } catch (error) {

      Logger.log(
        "エラー：" +
        fileId +
        " / " +
        error.message
      );
    }

  });


  SpreadsheetApp.flush();
}
